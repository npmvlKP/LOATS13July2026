    This function should be called at application startup to:
    1. Configure logging
    2. Initialize settings
    3. Set up any required directories
    """
    # Configure logging
    if os.environ.get("ENVIRONMENT") == "test":
        configure_logging(test_mode=True)
    else:
        configure_logging()
